    generatePDF(target) {
        if (!window.jspdf) {
            alert("Library jsPDF belum dimuat.");
            return;
        }

        const { jsPDF } = window.jspdf;
        const doc = new jsPDF({
            orientation: 'landscape',
            unit: 'mm',
            format: 'a4'
        });

        const state = window.TL.State.data;
        const isLulus = state.skor >= 9;
        const nama = state.nama || "Siswa";
        const kelas = state.kelas || "Kelas";
        const skor = state.skor || 0;

        const bgImg = new Image();
        bgImg.src = 'assets/images/bg-sertifikat.jpg';
        
        bgImg.onload = () => {
            doc.addImage(bgImg, 'JPEG', 0, 0, 297, 210);
            this.drawText(doc, isLulus, nama, kelas, skor);
            this.finishPDF(doc, target, isLulus);
        };
        
        bgImg.onerror = () => {
            doc.setFillColor(15, 23, 41);
            doc.rect(0, 0, 297, 210, 'F');
            doc.setDrawColor(245, 158, 66);
            doc.setLineWidth(2);
            doc.rect(10, 10, 277, 190);
            this.drawText(doc, isLulus, nama, kelas, skor);
            this.finishPDF(doc, target, isLulus);
        };
    },

    drawText(doc, isLulus, nama, kelas, skor) {

# PROMPT KONTEKS UTAMA: PROYEK TransforLab (Media Pembelajaran Interaktif Transformasi Geometri)

## 1. IDENTITAS DAN TUJUAN PROYEK
Anda adalah AI yang membantu pengembangan media pembelajaran interaktif bernama **TransforLab** (Laboratorium Penemuan Transformasi Geometri).
- **Target Pengguna:** Siswa SMP/MTs Fase D (Kelas IX) Kurikulum Merdeka.
- **Materi Pokok:** Transformasi Geometri (Refleksi, Translasi, Rotasi, Dilatasi, dan Matriks).
- **Platform:** Berbasis Web (browser desktop dan mobile), dibangun dengan **Vanilla HTML5, CSS3, dan JavaScript murni** (TANPA framework seperti React, Vue, atau Node.js). Cukup buka file `index.html` langsung di browser untuk menjalankannya.
- **Pendekatan Pedagogis:** Konstruktivisme & Guided Discovery Learning (siswa menemukan sendiri sifat transformasi melalui eksperimen interaktif, bukan dijejali rumus). Menggunakan siklus **PMDCF** (Predict, Manipulate, Discover, Compare, Formalize).

## 2. FITUR UTAMA & ARSITEKTUR TEKNIS
- **Struktur:** Menggunakan pola *Single Page Application (SPA)* murni tanpa framework. Semua navigasi dilakukan dengan fungsi JS `navigate(screenId)` yang mengganti konten `<div id="app">`.
- **Namespace Global:** Menggunakan `window.TL = {}` untuk semua modul JavaScript.
- **State Management:** Menggunakan `localStorage` untuk menyimpan nama siswa, progress modul, dan skor asesmen.
- **Library Eksternal (dengan Fallback Offline):**
  - **Three.js (`lib/three.min.js`)** untuk mode visual 3D (Bangun ruang, bidang cermin, sumbu rotasi).
  - **jsPDF (`lib/jsPDF.min.js`)** untuk generate Sertifikat PDF di sisi klien.
- **Struktur Folder:**
  - `index.html` (Entry point)
  - `css/` (`main.css`, `narasi.css`, `eksplorasi.css`, `assessment.css`)
  - `js/` (`app.js`, `state.js`, `narasi.js`, `form.js`, `dashboard.js`, `materi.js`, `canvas2d.js`, `canvas3d.js`, `temuan.js`, `assessment.js`, `certificate.js`)
  - `assets/` (Folder berisi gambar background, ikon, dan `data/soal.json`)
  - `lib/` (Folder berisi library Three.js dan jsPDF)

## 3. ALUR NARASI DAN UI (13 TAMPILAN UTAMA)
Media ini dibungkus dalam narasi **"Studio Arsitek Pola"**.
- **Tampilan 1 (Landing):** Halaman pembuka dengan tombol "START" (efek glow hover).
- **Tampilan 2–6 (Alur Narasi Latar):** Cerita seorang siswa magang yang sedang mencari kerja. Dia diterima magang di Studio Arsitek Pola, namun harus mempelajari 4 kemampuan dasar transformasi geometri. (Menggunakan background gambar statis beresolusi tinggi + popup *speech bubble* dari karakter Kak Sena di kiri bawah).
- **Tampilan 7 (Form Identitas):** Input Nama dan Kelas siswa untuk personalisasi media.
- **Tampilan 8 (Dashboard):** Menampilkan 4 folder besar: Refleksi, Translasi, Rotasi, Dilatasi. Dilengkapi *progress bar* (0/4) dan status badge (Done/Not Done) untuk setiap folder.
- **Tampilan 9 (Modul Eksplorasi):** Tampilan *Split-panel* (Canvas di kiri 65%, Sidebar di kanan 35%). Menu sidebar berisi 3 tab:
  1. **Materi:** Teks penjelasan dan gambar ilustrasi berdasarkan referensi buku matematika SMA.
  2. **Eksplorasi (Canvas 2D / 3D):** Alat eksperimen interaktif (drag bangun, slider skala, handle rotasi, drag garis cermin/vektor).
  3. **Temuan & Simpulan:** Pertanyaan pemandu, kotak jawaban siswa, dan rangkuman formal.
- **Tampilan 10 (Transisi Uji Kompetensi):** Narasi bahwa siswa siap diuji.
- **Tampilan 11 (Asesmen):** 12 Soal Pilihan Ganda bertingkat (C1–C6) dengan sistem *scoring*.
- **Tampilan 12 (Hasil Akhir):** Halaman LULUS (skor >= 70%) atau BELUM LULUS.
- **Tampilan 13 (Sertifikat):** Generate sertifikat PDF, memuat nama siswa, skor, dan gelar digital (Arsitek Muda / Peserta Magang).

## 4. DETAIL MODUL EKSPLORASI (4 MODUL + 1 KONEKSI)
*Filosofi: Eksplorasi dilakukan di atas bidang Kartesius (grid) secara interaktif.*
- **Modul 1 (Refleksi):** Percobaan garis cermin (terhadap sumbu X, Y, y=x, dan garis bebas). Mengamati tabel koordinat asal dan bayangan. Menemukan aturan perubahan tanda pada koordinat.
- **Modul 2 (Translasi):** Menggunakan vektor (panah) untuk menggeser bangun. Mengamati bahwa translasi tidak mengubah orientasi dan ukuran (isometri), serta jarak pergeseran tiap titik sama.
- **Modul 3 (Rotasi):** Menggunakan "handle busur derajat" untuk memutar bangun 90°, 180°, 270°. Membedakan rotasi di titik asal O(0,0) vs rotasi di titik pusat sembarang.
- **Modul 4 (Dilatasi):** Menggunakan slider interaktif untuk mengubah faktor skala (perbesaran: k>1, pengecilan: 0<k<1, pembalikan: k<0). Mengamati perbedaan antara transformasi Isometri (tidak mengubah ukuran) dan Non-Isometri (mengubah ukuran).
- **Modul 5 (Koneksi):** Dua refleksi terhadap garis sejajar setara dengan translasi. Dua refleksi terhadap garis berpotongan setara dengan rotasi (dengan sudut 2x sudut perpotongan).

## 5. GUIDELINE VISUAL & ASSET (STANDAR UNTUK GEMINI PRO)
Untuk menghasilkan aset grafis (background, avatar, ikon, sertifikat), gunakan gaya visual berikut:
- **Art Style:** Modern digital illustration, semi-realistis dengan garis bersih, *smooth shading*. Mirip gaya *Ghibli-inspired* bertemu ilustrasi editorial.
- **Palet Warna:** `Deep Navy (#0f1729, #1a2744)` untuk dasar, `Sky Blue (#38bdf8)`, `Amber (#f59e42)`, dan `Soft Golden (#ef3c7)` untuk aksen aural/cahaya hangat.
- **Karakter:** Pria muda Indonesia (remaja akhir/20-an awal), rambut hitam pendek, kulit sawo matang, kaos polos gelap. Ekspresi wajah bervariasi sesuai scene (netral, excited, kecewa).
- **Latar:** Kamar kerja sederhana dan rapi, meja kayu, laptop modern, lampu meja, jendela menghadap langit malam.
- **Ikon Folder:** Desain geometris minimalis di atas background gelap: Refleksi (segitiga & cermin), Translasi (panah geser), Rotasi (panah melingkar), Dilatasi (lingkaran konsentris).

3. KONVENSI KODE
Urutan load script di index.html:

lib/three.min.js, lalu lib/jspdf.min.js
Inline script kecil: cek if(!window.THREE) / if(!window.jspdf), kalau gagal document.write script tag CDN (three@0.160.0, jspdf@2.5.1) sebagai fallback

PENTING soal urutan: supaya urutan load TIDAK jadi sumber bug, app.js WAJIB memicu render pertama (TL.App.navigate('landing')) lewat window.addEventListener('load', ...), BUKAN dipanggil langsung di top-level script. Dengan begitu walau app.js dimuat lebih awal dari modul lain, render pertama baru benar-benar jalan setelah SEMUA script selesai dieksekusi — jadi TL.Narasi, TL.Dashboard, dst dijamin sudah ada saat dipanggil.
CSS dimuat di <head>: main.css dulu (base/reset), baru narasi.css dan eksplorasi.css (lebih spesifik, boleh override).
Namespace — tiap file JS menambahkan propertinya ke window.TL, contoh pola:

window.TL = window.TL || {};
window.TL.NamaModul = { /* ... */ };
Nama namespace yang WAJIB dipakai persis (jangan diganti-ganti antar file): TL.App, TL.State, TL.Narasi, TL.Form, TL.Dashboard, TL.Materi, TL.Canvas2D, TL.Canvas3D, TL.Temuan, TL.Assessment, TL.Certificate.
Render pattern: tiap "tampilan" = 1 fungsi yang generate HTML (template string atau DOM API) lalu dimasukkan ke <div id="app">. SPA murni, tidak ada page reload sama sekali.
Reuse komponen popup chat: narasi.js dipakai untuk T1-T6 dan T10, TAPI dialog T12A (lulus)/T12B (belum lulus)/T13 (menyerah) muncul di dalam alur assessment.js (setelah skor keluar). Supaya tidak duplikasi kode, narasi.js WAJIB mengekspos fungsi popup-chat generik yang bisa dipanggil dari file lain, misalnya TL.Narasi.renderChatBubbles(el, arrayDialog, onSelesai) — lalu assessment.js tinggal memanggil fungsi ini dengan naskah T12A/T12B/T13 (lihat bagian 6).
